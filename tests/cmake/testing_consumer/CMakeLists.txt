cmake_minimum_required(VERSION 3.20)
project(huxerui_testing_consumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
if (HUXERUI_TEST_PRELOAD_CORE)
    find_package(HuxerUI CONFIG REQUIRED)
endif ()
if (HUXERUI_TEST_LINK_FORM)
    find_package(HuxerUI CONFIG REQUIRED COMPONENTS testing ${HUXERUI_TEST_LINK_FORM})
else ()
    find_package(HuxerUI CONFIG REQUIRED COMPONENTS testing)
endif ()
get_target_property(testing_core HuxerUI::testing INTERFACE_LINK_LIBRARIES)
if (HUXERUI_TEST_LINK_FORM STREQUAL "shared" AND NOT testing_core STREQUAL "HuxerUI::huxerui")
    message(FATAL_ERROR "HuxerUI testing ignored the requested shared core")
elseif (HUXERUI_TEST_LINK_FORM STREQUAL "static" AND NOT testing_core STREQUAL "HuxerUI::huxerui_static")
    message(FATAL_ERROR "HuxerUI testing ignored the requested static core")
endif ()
find_package(HuxerUI CONFIG REQUIRED COMPONENTS testing)
if (HUXERUI_TEST_CONFLICT_LINK_FORM)
    find_package(HuxerUI CONFIG REQUIRED COMPONENTS testing ${HUXERUI_TEST_CONFLICT_LINK_FORM})
endif ()
add_executable(testing_consumer main.cpp)
target_link_libraries(testing_consumer PRIVATE HuxerUI::testing)
if (WIN32 AND testing_core STREQUAL "HuxerUI::huxerui")
    add_custom_command(TARGET testing_consumer POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                    "$<TARGET_FILE:HuxerUI::huxerui>" "$<TARGET_FILE_DIR:testing_consumer>"
            VERBATIM)
endif ()
enable_testing()
add_test(NAME InstalledTestingConsumer COMMAND testing_consumer)
